local function AvoidNextbot_GetHRP()
    local player = game.Players.LocalPlayer
    if not player or not player.Character then return nil end
    return player.Character:FindFirstChild("HumanoidRootPart")
end

local function AvoidNextbot_IsBotNearPosition(pos, radius)
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj:GetAttribute("Team") == "Nextbot" then
            local hrp = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("UpperTorso") or obj:FindFirstChild("Torso")
            if hrp and (hrp.Position - pos).Magnitude < radius then
                return true
            end
        end
    end
    return false
end

local function AvoidNextbot_FindClosestBot(hrp)
    local closestBot, closestDist = nil, math.huge
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") and obj:GetAttribute("Team") == "Nextbot" then
            local botRoot = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("UpperTorso") or obj:FindFirstChild("Torso")
            if botRoot then
                local d = (hrp.Position - botRoot.Position).Magnitude
                if d < closestDist then
                    closestDist = d
                    closestBot = { Model = obj, Position = botRoot.Position, Distance = d }
                end
            end
        end
    end
    return closestBot, closestDist
end

local function AvoidNextbot_GetPlayersTarget(botPos)
    local Players = game:GetService("Players")
    local LocalPlayer = Players.LocalPlayer
    local bestTarget, maxDist = nil, -1
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local pHrp = p.Character:FindFirstChild("HumanoidRootPart")
            if pHrp then
                local pDist = (botPos - pHrp.Position).Magnitude
                if pDist > maxDist then
                    maxDist = pDist
                    bestTarget = pHrp.Position
                end
            end
        end
    end
    if bestTarget then return bestTarget + Vector3.new(0, 3, 0) end
    return nil
end

local function AvoidNextbot_FindSafeLocations(centerPos)
    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = { game.Players.LocalPlayer.Character }

    local safeList = {}
    local attempts = 0
    while #safeList < 10 and attempts < 100 do
        attempts = attempts + 1
        local angle = math.rad(math.random(0, 360))
        local dist = math.random(30, 300)
        local offset = Vector3.new(math.cos(angle) * dist, 0, math.sin(angle) * dist)
        local checkPos = centerPos + offset
        local hit = workspace:Raycast(checkPos + Vector3.new(0, 50, 0), Vector3.new(0, -100, 0), rayParams)
        if hit and hit.Instance and hit.Instance.CanCollide then
            local groundPos = hit.Position
            if not AvoidNextbot_IsBotNearPosition(groundPos, 20) then
                table.insert(safeList, groundPos + Vector3.new(0, 3, 0))
            end
        end
    end
    return safeList
end

local function AvoidNextbot_FindSafeLocationsOnWall(wall)
    if not wall or not wall.Parent then return {} end
    local wallPart = wall:FindFirstChildWhichIsA("BasePart")
    if not wallPart then return {} end

    local wallPos = wallPart.Position
    local wallSize = wallPart.Size

    local safeList = {}
    local gridN = 8
    for i = -gridN, gridN do
        for j = -gridN, gridN do
            local offsetX = (i / gridN) * (wallSize.X / 2 - 5)
            local offsetZ = (j / gridN) * (wallSize.Z / 2 - 5)
            local candidatePos = Vector3.new(
                wallPos.X + offsetX,
                wallPos.Y + 2,
                wallPos.Z + offsetZ
            )
            if not AvoidNextbot_IsBotNearPosition(candidatePos, 20) then
                table.insert(safeList, candidatePos)
            end
        end
    end
    return safeList
end

local function AvoidNextbot_ApplyTeleport(botPos, hrp)
    local mode = Config.AvoidNextbotMode
    local targetPos = nil

    if mode == "Players" then
        targetPos = AvoidNextbot_GetPlayersTarget(botPos)
        if not targetPos then
            hrp.CFrame = hrp.CFrame + Vector3.new(0, 20, 0)
            return
        end
        if not AvoidNextbot_IsHiding then AvoidNextbot_OriginalPos = hrp.Position end
    elseif mode == "Wall_Safe" then
        if not wallModel or not wallModel.Parent then
            if CreateWall then CreateWall() end
            task.wait(0.3)
            if not wallModel or not wallModel.Parent then return end
        end
        if not AvoidNextbot_IsHiding then
            AvoidNextbot_OriginalPos = hrp.Position
        end
        if #AvoidNextbot_SafeLocations == 0 then
            AvoidNextbot_SafeLocations = AvoidNextbot_FindSafeLocationsOnWall(wallModel)
        end
        if #AvoidNextbot_SafeLocations > 0 then
            local bestIdx, bestDist = 1, -1
            for idx, loc in ipairs(AvoidNextbot_SafeLocations) do
                local d = (loc - botPos).Magnitude
                if d > bestDist then bestDist = d; bestIdx = idx end
            end
            targetPos = AvoidNextbot_SafeLocations[bestIdx]
            table.remove(AvoidNextbot_SafeLocations, bestIdx)
        end
        AvoidNextbot_IsHiding = true
    elseif mode == "Designated location" then
        if not AvoidNextbot_IsHiding then AvoidNextbot_OriginalPos = hrp.Position end
        if #AvoidNextbot_SafeLocations == 0 then
            AvoidNextbot_SafeLocations = AvoidNextbot_FindSafeLocations(hrp.Position)
        end
        if #AvoidNextbot_SafeLocations > 0 then
            local bestIdx, bestDist = 1, -1
            for idx, loc in ipairs(AvoidNextbot_SafeLocations) do
                local d = (loc - botPos).Magnitude
                if d > bestDist then bestDist = d; bestIdx = idx end
            end
            targetPos = AvoidNextbot_SafeLocations[bestIdx]
            table.remove(AvoidNextbot_SafeLocations, bestIdx)
        else
            targetPos = hrp.Position + Vector3.new(0, 20, 0)
        end
    end

    if targetPos then
        hrp.CFrame = CFrame.new(targetPos)
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    end
end

local function AvoidNextbot_RestorePosition()
    if AvoidNextbot_IsHiding and AvoidNextbot_OriginalPos then
        local hrp = AvoidNextbot_GetHRP()
        if hrp then
            hrp.CFrame = CFrame.new(AvoidNextbot_OriginalPos)
            hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        end
    end
    AvoidNextbot_IsHiding = false
    AvoidNextbot_OriginalPos = nil
    AvoidNextbot_OnWallPos = nil
end

                        if not AvoidNextbot_IsBotNearPosition(loc, 20) then
                            table.insert(AvoidNextbot_SafeLocations, loc)
                        end
                    end
                end
            end
        end
        for i = #AvoidNextbot_SafeLocations, 1, -1 do
            if AvoidNextbot_IsBotNearPosition(AvoidNextbot_SafeLocations[i], 20) then
                table.remove(AvoidNextbot_SafeLocations, i)
            end
        end
    end
end)

        end
    end
end)

    local billboard, label = createLabel(head, "Player", Color3.fromRGB(0, 255, 0))
    espPlayerObjects[targetPlayer] = { highlight = highlight, billboard = billboard, label = label, char = char }
end

    while task.wait(0.5) do
        if espPlayersEnabled then
            for _, plr in ipairs(game.Players:GetPlayers()) do
                if plr ~= game.Players.LocalPlayer then
                    if plr.Character and not espPlayerObjects[plr] then
                        createPlayerESP(plr)
                    end
                    if espPlayerObjects[plr] and espPlayerObjects[plr].char then
                        local obj = espPlayerObjects[plr]
                        local hum = obj.char:FindFirstChildOfClass("Humanoid")
                        local hrp = obj.char:FindFirstChild("HumanoidRootPart")
                        local myHrp = game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                        
                        if hrp and myHrp then
                            local dist = math.floor((hrp.Position - myHrp.Position).Magnitude)
                            obj.label.Text = string.format("%s | %dm", plr.Name, dist)

local function SaveClickLocation(x, y)
    EnsureFolder()
    pcall(function() writefile(LOCATION_FILE, tostring(x) .. "," .. tostring(y)) end)
end

local function LoadClickLocation()
    if type(isfile) == "function" and isfile(LOCATION_FILE) then
        local success, content = pcall(readfile, LOCATION_FILE)
        if success and content then
            local x, y = string.match(content, "([^,]+),([^,]+)")
            if x and y then return tonumber(x), tonumber(y) end
        end
    end
    return nil, nil
end

    if type(delfile) == "function" and type(isfile) == "function" and isfile(LOCATION_FILE) then
        pcall(function() delfile(LOCATION_FILE) end)
        return true
    end
    return false
end

local function SimulateKeyPress(keyCode)
    pcall(function()

        VIM:SendKeyEvent(true, keyCode, false, game)

local function DoAntiAfkAction()
    local action = math.random(1, 3)
    if action == 1 then
        SimulateKeyPress(Enum.KeyCode.Tab); task.wait(0.5); SimulateKeyPress(Enum.KeyCode.Tab)
    elseif action == 2 then
        SimulateKeyPress(Enum.KeyCode.W); task.wait(0.2)
    elseif action == 3 then
        SimulateKeyPress(math.random(1, 2) == 1 and Enum.KeyCode.C or Enum.KeyCode.F)
    end
end

    if antiAfkLoopRunning or not antiAfkEnabled then return end

    end)
end

local function GetValidItemNames(list)
    local names = {}
    for _, name in ipairs(list) do
        if type(name) == "string" and name ~= "" and name ~= "none" and name ~= "nil" then table.insert(names, name) end
    end
    return names
end

local function GetValidEventItemNames() return GetValidItemNames(EventItems) end
local function GetValidSpecialItemNames() return GetValidItemNames(SpecialEvent) end

local function HasAnyValidEventItem() return #GetValidEventItemNames() > 0 end
local function HasAnyValidSpecialItem() return #GetValidSpecialItemNames() > 0 end

local function FormatTargetItemsDisplay()
    local evtNames = GetValidEventItemNames()
    local spcNames = GetValidSpecialItemNames()
    local allNames = {}
    for _, v in ipairs(evtNames) do table.insert(allNames, v) end
    for _, v in ipairs(spcNames) do table.insert(allNames, v) end
    if #allNames == 0 then return "—" end
    return table.concat(allNames, ", ")
end

            lineNum = lineNum + 1
        end
    end
    for _, name in ipairs(SpecialEvent) do
        if type(name) == "string" and name ~= "" and name ~= "none" and name ~= "nil" then
            table.insert(lines, string.format("%d. %s | SpecialEvent [%s]", lineNum, name, string.format("%02d", itemCollectedCounts[name] or 0)))
            lineNum = lineNum + 1
        end
    end
    return #lines == 0 and "❌ Không có item nào" or table.concat(lines, "\n")

local function getAllItems()
    local items, player = {}, game.Players.LocalPlayer
    if not player then return items end
    local pc = player.Character
    local evtNames, spcNames = GetValidEventItemNames(), GetValidSpecialItemNames()
    if #evtNames == 0 and #spcNames == 0 then return items end

    local exactEventNames = {}
    local exactSpecialNames = {}
    for _, n in ipairs(evtNames) do exactEventNames[string.lower(n)] = n end
    for _, n in ipairs(spcNames) do exactSpecialNames[string.lower(n)] = n end

    local desc = workspace:GetDescendants()
    for i = 1, #desc do
        local v = desc[i]
        if (v:IsA("BasePart") or v:IsA("Model")) and not (pc and v:IsDescendantOf(pc)) then
            local rawName = v.Name
            local lowerName = string.lower(rawName)

            local matchedName, isSpecial = nil, false
            if exactEventNames[lowerName] then
                matchedName = exactEventNames[lowerName]
                isSpecial = false
            elseif exactSpecialNames[lowerName] then
                matchedName = exactSpecialNames[lowerName]
                isSpecial = true
            end

            if not matchedName then
                local baseName = string.match(lowerName, "^(%a+)%s*%(?%d*%)?$")
                if baseName then
                    if exactEventNames[baseName] then
                        matchedName = exactEventNames[baseName]
                        isSpecial = false
                    elseif exactSpecialNames[baseName] then
                        matchedName = exactSpecialNames[baseName]
                        isSpecial = true
                    end
                end
            end

            if matchedName then
                local part = v:IsA("BasePart") and v or v:FindFirstChildWhichIsA("BasePart")
                if part and part.Parent then
                    table.insert(items, {
                        Object = part,
                        Name = matchedName,
                        RawName = rawName,
                        Position = part.Position,
                        IsSpecial = isSpecial
                    })
                end
            end
        end
    end
    return items
end

local function SaveCurrentPosition(mode)
    local player = game.Players.LocalPlayer
    if not player or not player.Character then return end
    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end
    if mode == "event" and not savedPosEventSet then savedPosEvent, savedPosEventSet = hrp.Position, true
    elseif mode == "lv" and not savedPosLvSet then savedPosLv, savedPosLvSet = hrp.Position, true
    elseif mode == "token" and not savedPosTokenSet then savedPosToken, savedPosTokenSet = hrp.Position, true end
end

local function TeleportToSavedPosition(mode)
    local savedPos = (mode == "event" and savedPosEvent) or (mode == "lv" and savedPosLv) or (mode == "token" and savedPosToken)
    if not savedPos then return end
    local player = game.Players.LocalPlayer
    if player and player.Character then
        local hrp = player.Character:FindFirstChild("HumanoidRootPart")
        if hrp then hrp.CFrame = CFrame.new(savedPos) end
    end
end

local function IsAtSavedPosition(mode)
    local savedPos = (mode == "event" and savedPosEvent) or (mode == "lv" and savedPosLv) or (mode == "token" and savedPosToken)
    if not savedPos then return true end
    local player = game.Players.LocalPlayer
    if not player or not player.Character then return false end
    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return false end
    return (hrp.Position - savedPos).Magnitude <= POSITION_TOLERANCE
end

local function ClearSavedPosition(mode)
    if mode == "event" then savedPosEvent, savedPosEventSet = nil, false
    elseif mode == "lv" then savedPosLv, savedPosLvSet = nil, false
    elseif mode == "token" then savedPosToken, savedPosTokenSet = nil, false end
end

        conflictNotified = false
    end
end

local function GetActiveMode()

local function OnCharacterDeadOrDowned()
    if isCharacterDead then return end
    isCharacterDead, isFarmBlocked = true, true
    if isFarming then SetFarmEventStatus("🕑 Character Dead/Downed") end
    if isFarmingLv then SetFarmLvStatus("🕑 Character Dead/Downed") end
    if isRevivingActive then SetFarmTokenStatus("🕑 Character Dead/Downed") end

    local mode = GetActiveMode()
    task.spawn(function()
        while isCharacterDead do
            local player = game.Players.LocalPlayer
            if player and player.Character then
                local hum = player.Character:FindFirstChildOfClass("Humanoid")
                if hum and hum.Health > 0 then
                    isCharacterDead, isFarmBlocked = false, false
                    task.wait(1.5)
                    if isFarming then
                        if not wallModel or not wallModel.Parent then SetFarmEventStatus("🔨 Đang tạo wall_safe"); if CreateWall then CreateWall() end; task.wait(2) end
                        SetFarmEventStatus("❗ Teleporting - Wall_Safe"); if TeleportToWall then TeleportToWall() end
                    end
                    if isFarmingLv then
                        if not wallModelLv or not wallModelLv.Parent then SetFarmLvStatus("🔨 Đang tạo wall_safe"); if CreateWallLv then CreateWallLv() end; task.wait(2) end
                        SetFarmLvStatus("❗ Teleporting - Wall_Safe"); if TeleportToWallLv then TeleportToWallLv() end
                    end
                    if isRevivingActive then
                        if not wallModel or not wallModel.Parent then SetFarmTokenStatus("🔨 Đang tạo wall_safe"); if CreateWall then CreateWall() end; task.wait(2) end
                        SetFarmTokenStatus("❗ Teleporting - Wall_Safe"); if TeleportToWall then TeleportToWall() end
                    end
                    break
                else
                    if mode and not IsAtSavedPosition(mode) then TeleportToSavedPosition(mode) end
                    task.wait(0.15)
                end
            else task.wait(0.2) end
        end
    end)
end

task.spawn(function()
    local wasDead = false
    while task.wait(0.3) do
        local player = game.Players.LocalPlayer
        if not player or not player.Character then
            if not wasDead then wasDead = true; OnCharacterDeadOrDowned() end
        else
            local hum = player.Character:FindFirstChildOfClass("Humanoid")
            if hum then
                local isDead = hum.Health <= 0
                if isDead and not wasDead then wasDead = true; OnCharacterDeadOrDowned()
                elseif not isDead and wasDead then wasDead = false; isCharacterDead = false end
            end
        end
    end
end)

local function FindDownedPlayer()
    local Players, lp = game:GetService("Players"), game.Players.LocalPlayer
    if not lp then return nil end
    local closest, closestDist = nil, math.huge
    local myPos = lp.Character and lp.Character:FindFirstChild("HumanoidRootPart") and lp.Character.HumanoidRootPart.Position
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= lp then
            local char = player.Character
            if char then
                local hum = char:FindFirstChildOfClass("Humanoid")
                if hum and (hum.Health <= 0 or hum:GetState() == Enum.HumanoidStateType.Dead) then
                    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso") or char:FindFirstChildWhichIsA("BasePart")
                    if root then
                        if myPos then
                            local d = (myPos - root.Position).Magnitude
                            if d < closestDist then closestDist = d; closest = player end
                        else if not closest then closest = player end end
                    end
                end
            end
        end
    end
    return closest
end

local function RescueModeWait(target, rootPart)
    if not target or not target.Character or not rootPart then return false end
    local lp2 = game.Players.LocalPlayer
    TeleportWithEmote(rootPart.Position, nil, false, false)
    local isRevived = false
    while not isRevived and isRevivingActive do
        if isCharacterDead or not target or not target.Character then break end
        local hum = target.Character:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health > 0 then isRevived = true; break end
        local curRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if not curRoot then break end
        local lpRoot = lp2.Character and lp2.Character:FindFirstChild("HumanoidRootPart")
        if lpRoot and (lpRoot.Position - curRoot.Position).Magnitude > 5 then TeleportWithEmote(curRoot.Position + (lpRoot.Position - curRoot.Position).Unit * 3, nil, false, false) end
        if isRevivingActive then PressKeyEOnce() end
        task.wait(0.1)
    end
    return isRevived
end

local function RescueModeFast(target)
    if not target or not target.Character then return false end
    local isRevived, loopCount, MAX_LOOPS = false, 0, 50
    while not isRevived and isRevivingActive and loopCount < MAX_LOOPS do
        if isCharacterDead or not target or not target.Character then break end
        loopCount = loopCount + 1
        local hum = target.Character:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health > 0 then isRevived = true; break end
        local curRoot = target.Character:FindFirstChild("HumanoidRootPart")
        if not curRoot then break end
        StartSpamE(); TeleportWithEmote(curRoot.Position, nil, false, false); task.wait(0.2); StopSpamE()
        if not wallModel then CreateWall() task.wait(0.3) end
        SetFarmTokenStatus("❗ Teleporting - Wall_Safe"); TeleportToWallSafe(); task.wait(0.15)
        if not target or not target.Character then break end
        local hum2 = target.Character:FindFirstChildOfClass("Humanoid")
        if not hum2 or hum2.Health > 0 then isRevived = true; break end
    end
    StopSpamE(); return isRevived
end

-- ============================================================
local function RevivePlayersLoop()
    if reviveLoopRunning then return end
    reviveLoopRunning = true

    SetFarmTokenStatus("🔨 Đang tạo wall_safe"); CreateWall(); task.wait(2)
    SetFarmTokenStatus("❗ Teleporting - Wall_Safe"); TeleportToWall(); task.wait(1)
    isOnWallAndWaiting = true

    while isRevivingActive do
        if isCharacterDead then task.wait(0.5)
        elseif isOnWallAndWaiting then
            if IsModeBlocked() then task.wait(1)
            else
                SetFarmTokenStatus("🔎 Downed player - Đang quét")
                local target = FindDownedPlayer()
                if target then
                    isOnWallAndWaiting = false
                    local rootPart = target.Character and (target.Character:FindFirstChild("HumanoidRootPart") or target.Character:FindFirstChild("UpperTorso") or target.Character:FindFirstChild("Torso") or target.Character:FindFirstChildWhichIsA("BasePart"))
                    if rootPart then
                        CreateTargetESP(target)
                        tokenCurrentTargetName = "@" .. target.Name
                        tokenCurrentTargetPos = rootPart.Position
                        if game.Players.LocalPlayer.Character and game.Players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart") then
                            tokenCurrentTargetDist = (game.Players.LocalPlayer.Character.HumanoidRootPart.Position - rootPart.Position).Magnitude
                        end
                        reviveTarget = target
                        SetFarmTokenStatus("👤 Revive player - @" .. target.Name)

                        local isRevived = (farmMode == "Cứu nhanh") and RescueModeFast(target) or RescueModeWait(target, rootPart)

                        if isRevived then
                            reviveSuccessCount = reviveSuccessCount + 1
                            tokenStatSuccess = tokenStatSuccess + 1
                            SetFarmTokenStatus("✅ Revived successfully - " .. target.Name)
                            MarkTargetESPSuccess(); task.wait(2)
                        else
                            tokenStatFail = tokenStatFail + 1
                            SetFarmTokenStatus("❌ Revived Failure - " .. (target and target.Name or ""))
                            ClearTargetESP()
                        end

                        ClearTargetESP(); StopSpamE()
                        if not isCharacterDead then SetFarmTokenStatus("❗ Teleporting - Wall_Safe"); TeleportToWallSafe() end
                        task.wait(1); isOnWallAndWaiting = true
                        tokenCurrentTargetName, tokenCurrentTargetPos = "—", nil
                    end
                else
                    SetFarmTokenStatus("🔎 Downed player - Đang quét"); task.wait(1)
                end
            end
        else task.wait(0.1) end
    end
    reviveLoopRunning, isOnWallAndWaiting = false, false
end

local function StartRevivePlayers()
    if isRevivingActive then return end
    isRevivingActive = true
    reviveTarget, isOnWallAndWaiting, reviveSuccessCount = nil, false, 0
    tokenStatSuccess, tokenStatFail, tokenStatDead = 0, 0, 0
    tokenCurrentTargetName, tokenCurrentTargetPos = "—", nil
    SaveCurrentPosition("token"); UpdateFarmConflict()

    SetFarmTokenStatus("⏳ Đang khởi động"); task.wait(2)
    SetFarmTokenStatus("🔨 Đang tạo wall_safe"); CreateWall(); task.wait(2)
    SetFarmTokenStatus("❗ Teleporting - Wall_Safe"); TeleportToWall(); task.wait(0.5)

    task.spawn(RevivePlayersLoop)
end

local function StopRevivePlayers()
    isRevivingActive, reviveLoopRunning, reviveTarget, isOnWallAndWaiting = false, false, nil, false
    UpdateFarmConflict(); StopSpamE(); ClearTargetESP(); ForceResetMovement()
    if savedPosToken and savedPosTokenSet then
        task.spawn(function() task.wait(0.3); TeleportToSavedPosition("token"); task.wait(0.3); ClearSavedPosition("token") end)
    end
end

local function FarmEventLoop()
    if not isFarming or IsModeBlocked() then 
        task.wait(1)
        if isFarming then FarmEventLoop() end
        return 
    end

    if not (HasAnyValidEventItem() or HasAnyValidSpecialItem()) then 
        SetFarmEventStatus("🔴 Dừng... - Hết event")
        task.wait(1)
        if isFarming then FarmEventLoop() end
        return 
    end

    local player = game.Players.LocalPlayer
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not char or not hrp then 
        SetFarmEventStatus("🟠 Đang xử lý... - Chờ spawn")
        task.wait(0.3)
        if isFarming then FarmEventLoop() end
        return 
    end
    if isCollecting then 
        task.wait(0.1)
        if isFarming then FarmEventLoop() end
        return 
    end

    SetFarmEventStatus("🔎 Scan Items - Đang quét")
    local items = getAllItems()
    farmEventTargetName = FormatTargetItemsDisplay()

    if #items == 0 then
        farmEventTargetPos = nil
        ClearEventItemESP()
        task.wait(0.4)
        if isFarming then FarmEventLoop() end
        return
    end

    local playerPos = hrp.Position
    table.sort(items, function(a, b) 
        return (playerPos - a.Position).Magnitude < (playerPos - b.Position).Magnitude 
    end)

    local target = items[1]
    farmEventTargetPos, farmEventTargetDist = target.Position, (playerPos - target.Position).Magnitude

    CreateEventItemESP(target.Object, target.RawName or target.Name)
    isCollecting = true

    if target.IsSpecial then
        SetFarmEventStatus("⚡ Special Chase - " .. (target.RawName or target.Name))

        local function CollectItem(itemObject)
            pcall(function()
                local RS = game:GetService("ReplicatedStorage")
                local collectId = itemObject.Parent:GetAttribute("Id") or itemObject:GetAttribute("Id") or "a19ac91bff904b7385e826fd6a23dc01"
                if RS and RS.Events and RS.Events.Collectibles then 
                    RS.Events.Collectibles.Invoke:InvokeServer(player, collectId, "Collect") 
                end
            end)
        end

        local isVanished = false
        local collectSuccess = false

        while not collectSuccess do
            if not target.Object or not target.Object.Parent then
                isVanished = true
                collectSuccess = true
                break
            end
            if isCharacterDead then break end

            local currentHRP = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if not currentHRP then task.wait(0.05); continue end
            
            currentHRP.CFrame = CFrame.new(target.Object.Position)
            currentHRP.AssemblyLinearVelocity = Vector3.new(0, 0, 0)

            CollectItem(target.Object)
            
            SetFarmEventStatus("⚡ Special Chase - Đang teleport liên tục")
            task.wait(0.05)
        end

        if isVanished then
            IncrementItemCount(target.Name)
            SetFarmEventStatus("✅ Special collected - " .. target.Name)
        end

        if not isCharacterDead then
            SetFarmEventStatus("❗ Teleporting - Wall_Safe")
            TeleportToWallSafe()
        end

        isCollecting = false
        ClearEventItemESP()
        task.wait(0.1)

    else
        SetFarmEventStatus("👤 Đang teleporting " .. (target.RawName or target.Name))
        TeleportWithEmote(target.Position, target.Object.CFrame, false, false)
        task.wait(0.1)

        pcall(function()
            local RS = game:GetService("ReplicatedStorage")
            local collectId = target.Object.Parent:GetAttribute("Id") or target.Object:GetAttribute("Id") or "a19ac91bff904b7385e826fd6a23dc01"
            if RS and RS.Events and RS.Events.Collectibles then 
                RS.Events.Collectibles.Invoke:InvokeServer(player, collectId, "Collect") 
            end
        end)

        local checkTimeout = 0
        local MAX_CHECK_TIMEOUT = 10
        while checkTimeout < MAX_CHECK_TIMEOUT do
            if not target.Object or not target.Object.Parent then
                break
            end
            if isCharacterDead then break end
            local currentHRP = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
            if currentHRP and target.Object and target.Object.Parent then
                currentHRP.CFrame = CFrame.new(target.Object.Position)
            end
            task.wait(0.05)
            checkTimeout = checkTimeout + 0.05
        end

        IncrementItemCount(target.Name)

        isCollecting = false

local function formatTime(sec)
    sec = math.floor(sec or 0)
    local m = math.floor(sec / 60)
    local s = sec % 60
    return string.format("%02d:%02d", m, s)
end

local function formatDate(isoString)
    if not isoString or isoString == "" then return "—" end
    local y, m, d = string.match(isoString, "^(%d+)-(%d+)-(%d+)")
    if y and m and d then
        return string.format("%s/%s/%s", d, m, y)
    end
    return "—"
end

local function estimateSize(durationSec)
    local bytes = durationSec * 16 * 1024
    if bytes >= 1024 * 1024 then
        return string.format("%.2f MB", bytes / (1024 * 1024))
    elseif bytes >= 1024 then
        return string.format("%.2f KB", bytes / 1024)
    end
    return bytes .. " B"
end

local function checkSound(assetId, waitSeconds)
    waitSeconds = waitSeconds or 7
    local tempSound = Instance.new("Sound")
    tempSound.SoundId = assetId
    tempSound.Volume = 0
    tempSound.Parent = workspace

    local waited = 0
    while waited < waitSeconds do
        task.wait(0.1)
        waited = waited + 0.1
    end

    local ok = tempSound.IsLoaded and tempSound.TimeLength > 0
    local reason = ""

    if not ok then
        if not tempSound.IsLoaded then
            reason = "Không load được sound (asset có thể bị private, không tồn tại, hoặc sai định dạng)"
        elseif tempSound.TimeLength <= 0 then
            reason = "Sound không có thời lượng (file bị hỏng hoặc định dạng không được Roblox hỗ trợ)"
        else
            reason = "Không xác định được lỗi"
        end
    end

    local timeLength = tempSound.TimeLength
    tempSound:Destroy()

    return ok, reason, timeLength
end

local function checkLocalFile(path, waitSeconds)
    waitSeconds = waitSeconds or 7
    local ok, assetId = pcall(getcustomasset, path)
    if not ok then
        return false, "Không tạo được asset từ file (file không tồn tại hoặc client không hỗ trợ)", 0
    end
    return checkSound(assetId, waitSeconds)
end

local function startFileInfoLoop(sound, filename)
    local nameOnly, ext = string.match(filename, "^(.-)(%.%w+)$")
    nameOnly = nameOnly or filename
    ext = ext or ""

    task.spawn(function()
        local waited = 0
        while sound.TimeLength <= 0 and waited < 3 do
            task.wait(0.1)
            waited = waited + 0.1
        end

        while sound and sound.Parent do
            local cur = sound.TimePosition or 0
            local total = sound.TimeLength or 0
            pcall(function()
                FileInfo:SetDesc(
                    "File name: " .. nameOnly .. " | " .. ext .. "\n" ..
                    "Time of music: " .. formatTime(cur) .. " - " .. formatTime(total) .. "\n" ..
                    "Memory of music: " .. (total > 0 and estimateSize(total) or "—")
                )
            end)
            task.wait(0.5)
        end
    end)
end

local function isSupported(filename)
    local lower = string.lower(filename)
    for _, ext in ipairs(SUPPORTED_EXTS) do
        if string.sub(lower, -#ext) == ext then return true end
    end
    return false
end

local function refreshMusicList()
    musicList = {}
    musicPathMap = {}
    local ok, files = pcall(listfiles, "")
    if ok and files then
        for _, filePath in ipairs(files) do
            local filename = string.match(filePath, "([^/\\]+)$") or filePath
            if isSupported(filename) then
                table.insert(musicList, filename)
                musicPathMap[filename] = filePath
            end
        end
    end
    if #musicList == 0 then
        table.insert(musicList, "Không tìm thấy tệp nhạc")
    end
    return musicList
end

local function stopCurrent()

        pcall(function() currentSound:Destroy() end)
        currentSound = nil
    end
    restoreGameSounds()

local function playSelected()
    if not selectedTrack or selectedTrack == "Không tìm thấy tệp nhạc" then return end
    local path = musicPathMap[selectedTrack]
    if not path then return end

    local ok, assetId = pcall(getcustomasset, path)
    if not ok then
        WindUI:Notify({
            Title = "❌ Checkout Sound",
            Content = "Không tải được file: " .. selectedTrack,
            Duration = 3,
        })
        return
    end

    stopCurrent()

    local sound = Instance.new("Sound")
    sound.SoundId = assetId
    sound.Volume = 0.5
    sound.Looped = false
    sound.Parent = workspace
    currentSound = sound

    muteGameSounds(sound)
    sound:Play()

    startFileInfoLoop(sound, selectedTrack)

    sound.Ended:Connect(function()
        if autoPlayEnabled and currentSound == sound then
            task.wait(0.3)
            if autoPlayEnabled then
                sound:Play()
            end
        else
            restoreGameSounds()
        end
    end)

    sound:GetPropertyChangedSignal("Playing"):Connect(function()
        if autoPlayEnabled and currentSound == sound and not sound.Playing then
            task.wait(1)
            if autoPlayEnabled and currentSound == sound and not sound.Playing then
                pcall(function() sound:Play() end)
            end
        elseif not autoPlayEnabled and currentSound == sound and not sound.Playing then
            restoreGameSounds()
        end
    end)
end

local checkingFile = false

local function updateFileInfo(filename, path)
    if not filename then
        FileInfo:SetDesc(
            "File name: — | —\n" ..
            "Time of music: 00:00 - 00:00\n" ..
            "Memory of music: —"
        )
        return
    end
    local nameOnly, ext = string.match(filename, "^(.-)(%.%w+)$")
    nameOnly = nameOnly or filename
    ext = ext or ""
    local totalTime = 0
    if path then
        local ok, assetId = pcall(getcustomasset, path)
        if ok then
            local tempSound = Instance.new("Sound")
            tempSound.SoundId = assetId
            tempSound.Parent = workspace
            local waited = 0
            while tempSound.TimeLength <= 0 and waited < 3 do
                task.wait(0.1)
                waited = waited + 0.1
            end
            totalTime = tempSound.TimeLength or 0
            tempSound:Destroy()
        end
    end
    FileInfo:SetDesc(
        "File name: " .. nameOnly .. " | " .. ext .. "\n" ..
        "Time of music: " .. formatTime(0) .. " - " .. formatTime(totalTime) .. "\n" ..
        "Memory of music: " .. (totalTime > 0 and estimateSize(totalTime) or "—")
    )
end

    checkingFile = true

    WindUI:Notify({

        Duration = 3,

    task.spawn(function()

        if not ok then
            WindUI:Notify({
                Title = "❌ Checkout Sound",
                Content = "Không phát được: " .. option .. "\nLý do: " .. reason,
                Duration = 3,
            })
            checkingFile = false
            return
        end

        local nameOnly, ext = string.match(option, "^(.-)(%.%w+)$")
        nameOnly = nameOnly or option
        ext = ext or ""
        FileInfo:SetDesc(
            "File name: " .. nameOnly .. " | " .. ext .. "\n" ..

    end)
end
